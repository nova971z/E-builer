Modifier get_web3() pour essayer les fallbacks si le RPC principal échoue.
Si rpc_url est le défaut (pas un custom env var), essayer chaque fallback :
```python
def get_web3(rpc_url=None, testnet=False):
    if not HAS_WEB3:
        return None, False

    if rpc_url is None:
        if testnet:
            urls_to_try = [GMX_RPC_TESTNET]
        else:
            custom = os.environ.get("GMX_RPC_URL")
            urls_to_try = [custom] if custom else GMX_RPC_FALLBACKS
    else:
        urls_to_try = [rpc_url]

    with _web3_lock:
        # Check cache first
        for url in urls_to_try:
            if url in _web3_instances:
                w3 = _web3_instances[url]
                try:
                    if w3.is_connected():
                        return w3, True
                except Exception:
                    pass
                del _web3_instances[url]

        # Try connecting
        for url in urls_to_try:
            try:
                w3 = Web3(Web3.HTTPProvider(url, request_kwargs={"timeout": 10}))
                w3.middleware_onion.inject(ExtraDataToPOAMiddleware, layer=0)
                if w3.is_connected():
                    _web3_instances[url] = w3
                    log.info("Web3 connected: %s (chain=%d)", url, w3.eth.chain_id)
                    return w3, True
            except Exception as e:
                log.warning("Web3 RPC failed %s: %s", url, e)
                continue

    return None, False
```
