def fetch_binance(endpoint, params=None, ttl=10, timeout=5):
    cache_key = f"binance:{endpoint}:{json.dumps(params or {}, sort_keys=True)}"
    cached = cache.get(cache_key, ttl=ttl)
    if cached is not None:
        return cached

    for base_url in BINANCE_ENDPOINTS:
        try:
            resp = _http_session.get(base_url + endpoint, params=params, timeout=timeout)
            if resp.status_code == 200:
                data = resp.json()
                cache.set(cache_key, data)
                return data
            if resp.status_code == 403:
                continue  # essayer le prochain endpoint
        except Exception:
            continue

    log.warning("All Binance endpoints failed for %s", endpoint)
    return None
